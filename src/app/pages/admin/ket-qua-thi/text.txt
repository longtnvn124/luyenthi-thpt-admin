  loadCathi(status: string) {
    this.showAllTeam = true;
    let condition = this.httpParamsHelper.paramsConditionBuilder([
      // { conditionName: 'status', condition: IctuQueryCondition.equal, value: status },
      {
        conditionName: "status",
        condition: IctuQueryCondition.notEqual,
        value: "4",
        orWhere: "and",
      },
      {
        conditionName: "status",
        condition: IctuQueryCondition.notEqual,
        value: "1",
        orWhere: "and",
      },
      {
        conditionName: "status",
        condition: IctuQueryCondition.notEqual,
        value: "0",
        orWhere: "and",
      },
      {
        conditionName: "status",
        condition: IctuQueryCondition.notEqual,
        value: "-1",
        orWhere: "and",
      },
      // { conditionName: 'user_id', condition: IctuQueryCondition.equal, value: this.userId.toString(), orWhere: 'and' }
    ]);
    if (Number(status) === 3) {
      condition = this.httpParamsHelper.paramsConditionBuilder([
        {
          conditionName: "status",
          condition: IctuQueryCondition.equal,
          value: status,
        },
        // { conditionName: 'user_id', condition: IctuQueryCondition.equal, value: this.userId.toString(), orWhere: 'and' }
      ]);
    }
    forkJoin([
      this.tnQLCathiService.getTnQLCathiByCols(condition),
      this.userService.getUserByCol("role_ids", "2,6,7"),
    ]).subscribe(([res, teacher]) => {
      this.teacherFull = teacher.filter(
        (m) => m.status !== -1 && m.role_ids.findIndex((i) => i == "1") === -1
      );
      this.teacherTabletop = [...this.teacherFull];
      if (res && res.length !== 0) {
        this.reset();
        const tmpCathi = [];
        res.forEach((m) => {
          if (m.type_bank === "THI") {
            /* update */
            const d = new Date(m.time_start);
            const dateString = d.toLocaleString("en-GB");
            m["time_start_format"] = dateString;
            tmpCathi.push(m);
          }
        });
        this.dmCathiFull = this.helperService.sort(tmpCathi, "id", -1);
        this.dmCathi = [...this.dmCathiFull];
      } else {
        this.reset();
      }
    });
  }

  dataByRoom(f, i) {
    const ho = f["Hoten"]
      .split(" ")
      .splice(0, f["Hoten"].split(" ").length - 1)
      .join(" ");
    const ten = f["Hoten"]
      .split(" ")
      .splice(f["Hoten"].split(" ").length - 1, 1)
      .join(" ");
    let number_string = "";
    if (!f.bothi) {
      const readnumber = f["total"].toString().replace(/\D/gi, "|").split("|");
      number_string =
        readnumber.length < 2
          ? this.read_number.nguyen[readnumber[0].toString()]
          : this.read_number.nguyen[readnumber[0].toString()].concat(
              Number(readnumber[1]) !== 5 ? " phẩy " : " ",
              this.read_number.nguyen[readnumber[1].toString()]
            );
    }
    return [
      i,
      f["code"],
      f["student_code"],
      ho,
      ten,
      f["ngaysinh"],
      "",
      !f["bothi"] ? f["total"].toString().replace(/\D/gi, ",") : f["total"],
      number_string ? number_string : "",
      "",
    ];
  }
